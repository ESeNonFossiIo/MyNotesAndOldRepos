\amigurumi{Teddy Bear}{teddy-bear.jpg}

\begin{multicols}{2}

\subsection*{1. Head (Brown)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item *2 dc, inc* x 6 (24)
    \item *3 dc, inc* x 6 (30)
    \item *4 dc, inc* x 6 (36)
    \item *5 dc, inc* x 6 (42)
    \item \textbf{Rnds 8--13:} dc around (42)
    \item *5 dc, dec* x 6 (36)
    \item *4 dc, dec* x 6 (30)
    \item *3 dc, dec* x 6 (24)
    \item *2 dc, dec* x 6 (18)
    \item *1 dc, dec* x 6 (12)
    \item 6 dec (6)
\end{enumerate}
\textit{Stuff firmly. Insert safety eyes between rnds 10--11, about 7 sts apart.}

\subsection*{2. Body (Brown)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item *2 dc, inc* x 6 (24)
    \item *3 dc, inc* x 6 (30)
    \item *4 dc, inc* x 6 (36)
    \item \textbf{Rnds 7--12:} dc around (36)
    \item *4 dc, dec* x 6 (30)
    \item *3 dc, dec* x 6 (24)
    \item \textbf{Rnds 15--16:} dc around (24)
    \item *2 dc, dec* x 6 (18)
    \item *1 dc, dec* x 6 (12)
\end{enumerate}
\textit{Stuff as you go. Sew head to body, attaching at the narrowing rnd.}

\subsection*{3. Muzzle (Cream)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item \textbf{Rnds 4--5:} dc around (18)
\end{enumerate}
\textit{Embroider nose and mouth in black before sewing to face.}

\subsection*{4. Ears (Make 2, Brown)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item \textbf{Rnds 3--4:} dc around (12)
\end{enumerate}
\textit{Do not stuff. Flatten and sew to top of head.}

\subsection*{5. Inner Ears (Make 2, Cream)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item \textbf{Rnd 2:} dc around (6)
\end{enumerate}
\textit{Flatten and sew to the centre of each brown ear.}

\subsection*{6. Arms (Make 2, Brown)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item \textbf{Rnds 3--9:} dc around (12)
    \item 6 dec (6)
\end{enumerate}
\textit{Lightly stuff. Sew to sides of body just below the head.}

\subsection*{7. Legs (Make 2, Brown)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item \textbf{Rnds 4--9:} dc around (18)
    \item *1 dc, dec* x 6 (12)
\end{enumerate}
\textit{Stuff firmly. Sew to the bottom of the body so the bear can sit.}

\subsection*{8. Paw Pads (Make 2, Cream)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
\end{enumerate}
\textit{Sew to the underside of each foot before attaching legs.}

\subsection*{9. Bow Tie (Optional, Red)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item Chain 12, join with sl st to form a loop.
    \item \textbf{Rnds 2--4:} dc around (12)
    \item Pinch the centre and wrap a short length of yarn tightly around it.
\end{enumerate}
\textit{Slip onto the neck before sewing head and body together.}

\end{multicols}
