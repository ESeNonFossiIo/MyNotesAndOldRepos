\begin{multicols}{2}

\subsection*{1. Head (Green)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item *2 dc, inc* x 6 (24)
    \item *3 dc, inc* x 6 (30)
    \item *4 dc, inc* x 6 (36)
    \item \textbf{Rnds 7--11:} dc around (36)
    \item *4 dc, dec* x 6 (30)
    \item *3 dc, dec* x 6 (24)
    \item *2 dc, dec* x 6 (18)
    \item *1 dc, dec* x 6 (12)
    \item 6 dec (6)
\end{enumerate}
\textit{Fasten off. Insert safety eyes between rnds 9--10.}

\subsection*{2. Body (Green)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item *2 dc, inc* x 6 (24)
    \item *3 dc, inc* x 6 (30)
    \item \textbf{Rnds 6--10:} dc around (30)
    \item *3 dc, dec* x 6 (24)
    \item *2 dc, dec* x 6 (18)
    \item *1 dc, dec* x 6 (12)
    \item 6 dec (6)
\end{enumerate}

\subsection*{3. Nose / Muzzle (Green)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item *2 dc, inc* x 6 (24)
    \item \textbf{Rnds 5--7:} dc around (24)
\end{enumerate}

\subsection*{4. Flat Belly Patch (Cream)}
\textit{Worked in rows around a foundation chain.}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item Chain 6.
    \item Start in 2nd ch from hook: 4 dc, 3 dc in the last chain.
    \item Turn and work down opposite side of chain: 3 dc, 1 inc in final st. (12)
    \item Fasten off, leave long tail to sew onto the tummy.
\end{enumerate}

\subsection*{5. Tail (Green)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item *1 dc, inc* x 6 (18)
    \item \textbf{Rnd 4:} dc around (18)
    \item *2 dc, inc* x 6 (24)
    \item \textbf{Rnds 6--8:} dc around (24)
\end{enumerate}
\textit{Stuff base firmly to help Dino sit upright.}

\subsection*{6. Legs (Make 2)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item 6 inc (12)
    \item \textbf{Rnds 3--4:} dc around (12)
\end{enumerate}

\subsection*{7. Arms (Make 2)}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 6 dc into MR (6)
    \item \textbf{Rnds 2--5:} dc around (6)
\end{enumerate}

\subsection*{8. Scallops / Spikes (Yellow)}
\textit{Make 4 or 5.}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item 4 dc into MR (4)
    \item 4 inc (8)
\end{enumerate}
\textit{Do not stuff. Flatten into triangles.}

\subsection*{9. Tiny Wings (Green)}
\textit{Make 2.}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item Chain 2.
    \item 2 dc in 2nd chain from hook, ch 1, turn.
    \item 2 inc (4)
\end{enumerate}
\textit{Fasten off. Sew to upper back.}
